`Container` wraps a core SnapSort container and provides it to descendant `Item` and nested `Container` components through React context.

React owns the rendered collection. Supply synchronous state and ghost
callbacks for every structural operation the container accepts; the adapter
does not fall back to SnapSort's Vanilla DOM callbacks.

## Define Container State

Keep persistent items and temporary ghosts in separate state collections.

```tsx
import { useCallback, useState } from "react";
import type {
  GhostInsertEvent,
  GhostRemoveEvent,
  ItemMoveEvent,
} from "@snap-engine/snapsort";

type Task = { id: string; label: string };
type Column = { id: string; title: string; items: Task[] };

const initialColumns: Column[] = [
  {
    id: "todo",
    title: "To do",
    items: [
      { id: "task-1", label: "Sketch layout" },
      { id: "task-2", label: "Wire behavior" },
    ],
  },
  { id: "done", title: "Done", items: [] },
];
```

## Define Callback Functions

Pure helpers keep the framework callbacks small and make their state changes
easy to test independently.

```tsx
function moveTask(current: Column[], event: ItemMoveEvent): Column[] {
  const itemId = String(event.itemId);
  const targetId = String(event.to.containerMetadata.columnId);
  const task = current
    .flatMap((column) => column.items)
    .find((entry) => entry.id === itemId);
  if (!task) return current;

  return current.map((column) => {
    const items = column.items.filter((entry) => entry.id !== itemId);
    if (column.id === targetId) {
      items.splice(Math.min(event.to.index, items.length), 0, task);
    }
    return { ...column, items };
  });
}

function upsertGhost(
  current: GhostInsertEvent[],
  event: GhostInsertEvent,
): GhostInsertEvent[] {
  return [
    ...current.filter((entry) => entry.ghostItem !== event.ghostItem),
    event,
  ];
}

function removeGhost(
  current: GhostInsertEvent[],
  event: GhostRemoveEvent,
): GhostInsertEvent[] {
  return current.filter((entry) => entry.ghostItem !== event.ghostItem);
}
```

Use those helpers inside the component callbacks. These state updates remain
synchronous because the adapter invokes them through `flushMutation`.

```tsx
function useTaskBoardState() {
  const [columns, setColumns] = useState(initialColumns);
  const [ghosts, setGhosts] = useState<GhostInsertEvent[]>([]);

  const onItemMove = useCallback((event: ItemMoveEvent) => {
    setColumns((current) => moveTask(current, event));
  }, []);

  const onGhostInsert = useCallback((event: GhostInsertEvent) => {
    setGhosts((current) => upsertGhost(current, event));
  }, []);

  const onGhostRemove = useCallback((event: GhostRemoveEvent) => {
    setGhosts((current) => removeGhost(current, event));
  }, []);

  return {
    columns,
    ghosts,
    callbacks: { onItemMove, onGhostInsert, onGhostRemove },
  };
}
```

## Merge Ghosts into Rendered Entries

`onGhostInsert` supplies a core index that excludes dragged items. Convert it
into the keyed entry list React will render.

```tsx
type RenderEntry =
  | { kind: "task"; task: Task }
  | { kind: "ghost"; event: GhostInsertEvent };

function renderColumnEntries(
  column: Column,
  ghosts: GhostInsertEvent[],
): RenderEntry[] {
  const entries: RenderEntry[] = column.items.map((task) => ({
    kind: "task",
    task,
  }));
  const relevantGhosts = ghosts
    .filter((event) => event.containerMetadata.columnId === column.id)
    .sort((a, b) => a.index - b.index);
  const draggedIds = new Set(
    relevantGhosts[0]?.itemIds.map(String) ?? [],
  );

  for (const event of relevantGhosts) {
    let persistentIndex = 0;
    let renderedIndex = 0;
    while (renderedIndex < entries.length && persistentIndex < event.index) {
      const entry = entries[renderedIndex];
      if (entry.kind === "task" && !draggedIds.has(entry.task.id)) {
        persistentIndex += 1;
      }
      renderedIndex += 1;
    }
    while (renderedIndex < entries.length) {
      const entry = entries[renderedIndex];
      if (entry.kind !== "task" || !draggedIds.has(entry.task.id)) break;
      renderedIndex += 1;
    }
    entries.splice(renderedIndex, 0, { kind: "ghost", event });
  }

  return entries;
}
```

## Define a Container

Render the keyed item and ghost entries owned by React.

```tsx
import { Container, Ghost, Item } from "@snap-engine/snapsort/react";
import type { ContainerCallbacks, GhostInsertEvent } from "@snap-engine/snapsort";

function TaskContainer({
  column,
  ghosts,
  callbacks,
}: {
  column: Column;
  ghosts: GhostInsertEvent[];
  callbacks: ContainerCallbacks;
}) {
  const renderedEntries = renderColumnEntries(column, ghosts);

  return (
    <Container
      metadata={{ columnId: column.id }}
      config={{ callbacks }}
    >
      {renderedEntries.map((entry) =>
        entry.kind === "ghost" ? (
          <Ghost key={`ghost-${entry.event.ghostItem.id}`} event={entry.event} />
        ) : (
          <Item key={entry.task.id} itemId={entry.task.id}>
            {entry.task.label}
          </Item>
        ),
      )}
    </Container>
  );
}
```

See the [Item reference](/docs/snapsort/reference/react/item) for extracting
the item branch into a reusable component.

## Props

```ts
type ContainerProps = {
  children?: ReactNode;
  className?: string;
  config?: ContainerConfig;
  containerObject?: SnapSortContainer | null;
  itemId?: string;
  locked?: boolean;
  selected?: boolean;
  metadata?: Record<string, unknown>;
  style?: CSSProperties;
};
```

| Prop | Default | Description |
|------|---------|-------------|
| `children` | `undefined` | Item and nested container content. |
| `className` | `""` | Extra class string applied to the rendered element. |
| `config` | `{}` | Optional SnapSort container configuration. |
| `containerObject` | `null` | Optional existing core container object. The component does not own its creation. |
| `itemId` | `undefined` | Required when this container is a sortable entry inside another container. |
| `locked` | `true` | Keeps a nested container from being dragged. Set to `false` to make it draggable. |
| `selected` | `false` | Consumer-owned selection flag for multi-item drags. Only meaningful when `locked` is `false`. |
| `metadata` | `{}` | Metadata assigned to the container object and exposed in callback events. |
| `style` | `undefined` | Inline styles merged onto the container element. |

`Container` forwards a ref to its core object and renders a `div` with `snapsort-container` plus a mode-specific class.

## Config

| Option | Default | Description |
|--------|---------|-------------|
| `mode` | `"euclidean"` | Drag/drop mode for this tree: `"euclidean"`, `"progressive"`, `"insertion"`, or `"swap"`. |
| `direction` | `"column"` | Main layout direction: `"column"` or `"row"`. |
| `mainAxisAlign` | `"start"` | Main-axis alignment for virtual placement. Supports `"start"` and `"center"`. |
| `name` | Generated | Human-readable container name. |
| `animation` | `undefined` | Optional reorder, drop, and click-move animation settings. Omit it or set it to `null` for no animation. |
| `dropPriority` | `0` | Base preference for candidate positions owned directly by this container. Higher eligible priorities are considered first. |
| `callbacks` | No persistent mutation default | Required framework-state/ghost callbacks plus lifecycle and validation hooks. |

Animations are opt-in. Use SnapSort's standard 100ms preset when you want the
library's conventional motion:

```tsx
import { defaultAnimations } from "@snap-engine/snapsort";

<Container config={{ animation: defaultAnimations }} />
```

Mode is resolved from the root container when a drag starts, so nested containers in one drag tree should use the same mode.

## Callbacks

Callbacks belong to the specific `Container` where they are configured. They
are never inherited by nested containers and never bubble to a parent. The
receiver shown below is the container whose `callbacks` object SnapSort reads;
event fields such as `source`, `from`, `to`, and `current` describe locations
and do not change that receiver.

| Callback | Fires on | Built-in modes / trigger | Notes |
|----------|----------|--------------------------|-------|
| `onItemMove(event)` | Direct destination (`event.to.container`) | Euclidean/progressive move or copy commits; insertion copy or placement-changing moves; placement-changing programmatic `moveItem`; Vanilla's swap fallback | Preferred state callback. It reports one semantic move (or copy insertion) with both `from` and `to`; same-placement insertion/programmatic moves are no-ops, and the source does not also receive `onItemRemove`. |
| `onItemInsert(event)` | Direct destination (`event.container`) | Primitive insertions; otherwise the move, copy, programmatic, and Vanilla swap-fallback paths when that destination has no `onItemMove` | Lower-level destination primitive and fallback. Configure either this callback or `onItemMove` anywhere an item can land. |
| `onItemRemove(event)` | Current direct owner, or transient clone host (`event.container`) | Programmatic `removeItem`; cleanup of an uncommitted euclidean/progressive copy clone | This is not the source half of an ordinary move. Programmatic events have `session === null`; copy cleanup carries its session. |
| `onItemSwap(event)` | Dragged item's pre-swap direct source (`event.a.container`) | Swap commit only | Receives both participants once. The target container does not receive a second swap callback. Current limitation: swap mode does not honor `dropEffect` values `"copy"` or `"none"`; it still follows the swap path. Required by the React adapter. |
| `onDragClone(event)` | Tree root | Euclidean/progressive start with `session.dropEffect === "copy"` | Runs inside the root's mutation transaction. Materialize and bind every `cloneItems` entry synchronously or the copy drag is vetoed. |
| `onDragStart(event)` | Tree root | Start of every mode | Runs directly, outside `flushMutation`. Return `false` to veto; `event.source.container` identifies the direct source. |
| `onDragEnd(event)` | Tree root | End of every activated drag, including cancellation | Runs through the root's `flushMutation` after the final mutation or return path. It does not fire after a startup veto or pre-activation failure. Inspect `event.source` and `event.destination` for locations. |
| `onDropTargetChange(event)` | Tree root | Every mode, when the prospective container or index changes | Runs through the root's `flushMutation`. `event.current?.container` is the prospective target, not the receiver. Flow copy must be chosen in `onDragStart` and must not be toggled here. |
| `onDragItemEnter(event)` | Direct owner of `event.overItem` (`event.container`) | Every mode, when the pointer first enters an item's hitbox | Semantically distinct from resolved slot changes, but hit-testing is scoped to items in the currently resolved target container. Runs directly. |
| `onDragItemMove(event)` | Direct owner of `event.overItem` (`event.container`) | Every mode, on each pointer move while still inside an item's hitbox | Semantically distinct from resolved slot changes, but hit-testing is scoped to items in the currently resolved target container. Runs directly. |
| `onDragItemLeave(event)` | Direct owner of `event.overItem` (`event.container`) | Every mode, when the pointer leaves an item's hitbox or the drag ends | Part of the same target-scoped hover stream; the final leave prevents hover UI from remaining active. Runs directly. |
| `onVisualGeometryInvalidated(event)` | Tree root (`event.root`) | Mode-independent, after transient drag, ghost, animation, or settle geometry changes | Direct notification, including outside an active session; coalesced to at most once per engine frame. |
| `canDrop(event)` | Each candidate destination (`event.container`) | Every mode, once per candidate container during each target resolution | Runs directly before priority. Return `false` to reject all candidates owned directly by that container for this resolution. Programmatic moves bypass it. |
| `getDropPriority(event)` | Each eligible candidate destination (`event.container`) | Every mode, once per eligible container during each target resolution | Runs directly after `canDrop`. Return a finite override or `undefined` to keep `dropPriority`. Programmatic moves bypass it. |
| `createGhost(event)` | Container where the ghost is first created (`event.container`) | Flow move/none: initial source; flow copy: first valid destination; insertion: first valid destination; swap: root | Creation notification only; moving an existing ghost does not fire it again. Returned DOM nodes are ignored because React owns ghost DOM. Runs directly. |
| `onGhostInsert(event)` | Current ghost owner (`event.container`) | Flow source/destination placement; insertion target placement; swap root pointer placement and movement | Required on every possible React ghost owner. Store/upsert the event and render `Ghost`; runs through that owner's `flushMutation`. |
| `onGhostRemove(event)` | Ghost owner it is leaving (`event.container`) | Ghost crosses containers, is replaced, or the session ends | Required on every possible React ghost owner. Remove the matching rendered ghost; runs through the departing owner's `flushMutation`. |
| `flushMutation(mutation)` | Same container as the callback being transacted | Structural and lifecycle mutations in every mode, plus programmatic mutations | Adapter hook, not a session event. React supplies it automatically. It wraps item move/insert/remove/swap, ghost insert/remove, and root drag clone, target-change, and drag-end callbacks. |
| `awaitMutation()` | Same receiver, only when `flushMutation` is absent | Deprecated compatibility path after a mutation | React supplies `flushMutation`, so this fallback is not used. Promise results cannot provide a paint-atomic commit. |

The table describes **dispatch**, which is separate from startup
**configuration validation**. For framework-owned trees, SnapSort verifies
that the source can service mutations before activating a drag even though a
successful move is later dispatched on the destination.

- In euclidean, progressive, and insertion trees, configure either
  `onItemMove` or `onItemInsert` on every possible direct source and
  destination.
- Configure `onItemSwap` on every container that can be the direct source of a
  swap.
- Configure `onGhostInsert` and `onGhostRemove` on every possible ghost owner:
  flow move/none sources and flow destinations; insertion sources (validated
  at startup) and destinations (actual receivers); and the root in swap mode.
- A shared handler object is fine, but pass it explicitly to every
  participating `Container`; nesting alone does not share callbacks.

Configure `onItemRemove` as well when a container can receive
`removeItem(id)` or can host a transient flow-copy clone. Ordinary moves emit
only the destination-owned `onItemMove`/`onItemInsert` event.

Update React state synchronously in `onItemMove` or `onItemSwap`. The adapter
wraps structural callbacks in `flushSync`, allowing SnapSort to read committed
DOM before paint. Do not defer these updates with transitions, timers,
animation frames, or promises. Missing callbacks fail clearly instead of
mutating framework-rendered DOM.

### Drop policy events

Drop targeting follows eligibility, priority, then the selected placement
mode. `CanDropEvent` includes `itemMetadata`/`itemsMetadata`, destination
`containerMetadata`, the primary `source`, and `sources` parallel to the full
multi-item drag. `DropPriorityEvent` adds `staticPriority`, pointer and frozen
rectangle geometry, and destination depth. Priority applies only to candidates
owned directly by this container and does not propagate to nested containers.

See [Drop Policies](/docs/snapsort/guides/04_drop_policies) for metadata-driven
grouping, exact resolution rules, and reusable callbacks from
`@snap-engine/snapsort/callbacks`.

## Imperative Ref

Container components forward refs to their core container object.

```tsx
import { useRef } from "react";
import { Container, Item } from "@snap-engine/snapsort/react";
import type { Container as SnapSortContainer } from "@snap-engine/snapsort";

export function ClickMove() {
  const source = useRef<SnapSortContainer>(null);
  const target = useRef<SnapSortContainer>(null);

  function moveToTarget(id: string) {
    if (!source.current || !target.current) return;
    source.current.moveItem(id, target.current, target.current.numberOfItems);
  }

  return (
    <>
      <Container ref={source} config={{ callbacks }}>
        {renderedSourceEntries}
      </Container>
      <Container ref={target} config={{ callbacks }}>
        {renderedTargetEntries}
      </Container>
    </>
  );
}
```

The omitted `callbacks` and rendered-entry construction are the same
state/ghost pattern shown in Setup. Imperative and pointer moves both commit
through `onItemMove`; `moveItem` never authorizes direct DOM mutation. It is an
authoritative application command and bypasses `canDrop` and priority policy.

The ref exposes `moveItem`, `removeItem`, `numberOfItems`, `dropPriority`,
`direction`, `mainAxisAlign`, `mode`, `config`, `callbacks`, and `dragSession`
from the core container.
