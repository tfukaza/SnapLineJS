`Container` wraps a core SnapSort container and provides it to descendant `Item` and nested `Container` components through React context.

React owns the rendered collection. Supply synchronous state and ghost
callbacks for every structural operation the container accepts; the adapter
does not fall back to SnapSort's Vanilla DOM callbacks.

## Define Container State

Keep persistent items and temporary ghosts in separate state collections.

```tsx
import { useCallback, useState } from "react";
import type {
  GhostInsertEvent,
  GhostRemoveEvent,
  ItemMoveEvent,
} from "@snap-engine/snapsort";

type Task = { id: string; label: string };
type Column = { id: string; title: string; items: Task[] };

const initialColumns: Column[] = [
  {
    id: "todo",
    title: "To do",
    items: [
      { id: "task-1", label: "Sketch layout" },
      { id: "task-2", label: "Wire behavior" },
    ],
  },
  { id: "done", title: "Done", items: [] },
];
```

## Define Callback Functions

Pure helpers keep the framework callbacks small and make their state changes
easy to test independently.

```tsx
function moveTask(current: Column[], event: ItemMoveEvent): Column[] {
  const itemId = String(event.itemId);
  const targetId = String(event.to.containerMetadata.columnId);
  const task = current
    .flatMap((column) => column.items)
    .find((entry) => entry.id === itemId);
  if (!task) return current;

  return current.map((column) => {
    const items = column.items.filter((entry) => entry.id !== itemId);
    if (column.id === targetId) {
      items.splice(Math.min(event.to.index, items.length), 0, task);
    }
    return { ...column, items };
  });
}

function upsertGhost(
  current: GhostInsertEvent[],
  event: GhostInsertEvent,
): GhostInsertEvent[] {
  return [
    ...current.filter((entry) => entry.ghostItem !== event.ghostItem),
    event,
  ];
}

function removeGhost(
  current: GhostInsertEvent[],
  event: GhostRemoveEvent,
): GhostInsertEvent[] {
  return current.filter((entry) => entry.ghostItem !== event.ghostItem);
}
```

Use those helpers inside the component callbacks. These state updates remain
synchronous because the adapter invokes them through `flushMutation`.

```tsx
function useTaskBoardState() {
  const [columns, setColumns] = useState(initialColumns);
  const [ghosts, setGhosts] = useState<GhostInsertEvent[]>([]);

  const onItemMove = useCallback((event: ItemMoveEvent) => {
    setColumns((current) => moveTask(current, event));
  }, []);

  const onGhostInsert = useCallback((event: GhostInsertEvent) => {
    setGhosts((current) => upsertGhost(current, event));
  }, []);

  const onGhostRemove = useCallback((event: GhostRemoveEvent) => {
    setGhosts((current) => removeGhost(current, event));
  }, []);

  return {
    columns,
    ghosts,
    callbacks: { onItemMove, onGhostInsert, onGhostRemove },
  };
}
```

## Merge Ghosts into Rendered Entries

`onGhostInsert` supplies a core index that excludes dragged items. Convert it
into the keyed entry list React will render.

```tsx
type RenderEntry =
  | { kind: "task"; task: Task }
  | { kind: "ghost"; event: GhostInsertEvent };

function renderColumnEntries(
  column: Column,
  ghosts: GhostInsertEvent[],
): RenderEntry[] {
  const entries: RenderEntry[] = column.items.map((task) => ({
    kind: "task",
    task,
  }));
  const relevantGhosts = ghosts
    .filter((event) => event.containerMetadata.columnId === column.id)
    .sort((a, b) => a.index - b.index);
  const draggedIds = new Set(
    relevantGhosts[0]?.itemIds.map(String) ?? [],
  );

  for (const event of relevantGhosts) {
    let persistentIndex = 0;
    let renderedIndex = 0;
    while (renderedIndex < entries.length && persistentIndex < event.index) {
      const entry = entries[renderedIndex];
      if (entry.kind === "task" && !draggedIds.has(entry.task.id)) {
        persistentIndex += 1;
      }
      renderedIndex += 1;
    }
    while (renderedIndex < entries.length) {
      const entry = entries[renderedIndex];
      if (entry.kind !== "task" || !draggedIds.has(entry.task.id)) break;
      renderedIndex += 1;
    }
    entries.splice(renderedIndex, 0, { kind: "ghost", event });
  }

  return entries;
}
```

## Define a Container

Render the keyed item and ghost entries owned by React.

```tsx
import { Container, Ghost, Item } from "@snap-engine/snapsort/react";
import type { ContainerCallbacks, GhostInsertEvent } from "@snap-engine/snapsort";

function TaskContainer({
  column,
  ghosts,
  callbacks,
}: {
  column: Column;
  ghosts: GhostInsertEvent[];
  callbacks: ContainerCallbacks;
}) {
  const renderedEntries = renderColumnEntries(column, ghosts);

  return (
    <Container
      metadata={{ columnId: column.id }}
      config={{ callbacks }}
    >
      {renderedEntries.map((entry) =>
        entry.kind === "ghost" ? (
          <Ghost key={`ghost-${entry.event.ghostItem.id}`} event={entry.event} />
        ) : (
          <Item key={entry.task.id} itemId={entry.task.id}>
            {entry.task.label}
          </Item>
        ),
      )}
    </Container>
  );
}
```

See the [Item reference](/docs/snapsort/reference/react/item) for extracting
the item branch into a reusable component.

## Props

```ts
type ContainerProps = {
  children?: ReactNode;
  className?: string;
  config?: ContainerConfig;
  containerObject?: SnapSortContainer | null;
  itemId?: string;
  locked?: boolean;
  selected?: boolean;
  metadata?: Record<string, unknown>;
  style?: CSSProperties;
};
```

| Prop | Default | Description |
|------|---------|-------------|
| `children` | `undefined` | Item and nested container content. |
| `className` | `""` | Extra class string applied to the rendered element. |
| `config` | `{}` | Optional SnapSort container configuration. |
| `containerObject` | `null` | Optional existing core container object. The component does not own its creation. |
| `itemId` | `undefined` | Required when this container is a sortable entry inside another container. |
| `locked` | `true` | Keeps a nested container from being dragged. Set to `false` to make it draggable. |
| `selected` | `false` | Consumer-owned selection flag for multi-item drags. Only meaningful when `locked` is `false`. |
| `metadata` | `{}` | Metadata assigned to the container object and exposed in callback events. |
| `style` | `undefined` | Inline styles merged onto the container element. |

`Container` forwards a ref to its core object and renders a `div` with `snapsort-container` plus a mode-specific class.

## Config

| Option | Default | Description |
|--------|---------|-------------|
| `mode` | `"euclidean"` | Drag/drop mode for this tree: `"euclidean"`, `"progressive"`, `"insertion"`, or `"swap"`. |
| `strategy` | Resolved from `mode` | Advanced custom `{ dropTarget, lifecycle }` pair that overrides `mode`; its resolver owns eligibility and priority policy. |
| `direction` | `"column"` | Main layout direction: `"column"` or `"row"`. |
| `mainAxisAlign` | `"start"` | Main-axis alignment for virtual placement. Supports `"start"` and `"center"`. |
| `name` | Generated | Human-readable container name. |
| `animation` | Default 100ms animations | Reorder, drop, and click-move animation settings. Set to `null` to disable configured animations. |
| `disableFlip` | `false` | Disables FLIP movement animation. |
| `dropPriority` | `0` | Base preference for candidate positions owned directly by this container. Higher eligible priorities are considered first. |
| `callbacks` | No persistent mutation default | Required framework-state/ghost callbacks plus lifecycle and validation hooks. |

Mode is resolved from the root container when a drag starts, so nested containers in one drag tree should use the same mode.

## Callbacks

| Callback | When it runs |
|----------|--------------|
| `onItemMove(event)` | An item moves to a new container or index. Preferred for state-backed lists because it includes both `from` and `to`. |
| `onItemSwap(event)` | Swap mode commits a pairwise slot exchange. Required for swap mode in the React adapter. |
| `onItemInsert(event)` | An item is inserted into a container; this is the lower-level fallback for `onItemMove`. |
| `onItemRemove(event)` | An item is removed through `removeItem(id)`. |
| `onDragStart(event)` | A drag is starting. Return `false` to veto it. |
| `onDragEnd(event)` | A drag has ended and its mutation has committed. |
| `onDropTargetChange(event)` | The prospective drop container or index changed. Fires on the root container. |
| `canDrop(event)` | Runs once for this destination before priority. Return `false` to exclude its directly owned candidates. |
| `getDropPriority(event)` | Runs once after eligibility. Return a finite priority override, or `undefined` to preserve `dropPriority`. |
| `onGhostInsert(event)` | A ghost item or insertion marker is inserted or moved. Required for interactive React containers; store the event and render `Ghost`. |
| `onGhostRemove(event)` | A ghost item or insertion marker is removed. Required for interactive React containers; remove the matching rendered ghost. |
| `createGhost(event)` | Notification hook only in React. Returned DOM nodes are ignored because React owns ghost DOM. |
| `flushMutation(mutation)` | Runs a structural callback inside React's synchronous commit. Supplied automatically. |
| `awaitMutation()` | Deprecated compatibility callback. |

Update React state synchronously in `onItemMove` or `onItemSwap`. The adapter
wraps structural callbacks in `flushSync`, allowing SnapSort to read committed
DOM before paint. Do not defer these updates with transitions, timers,
animation frames, or promises. Missing callbacks fail clearly instead of
mutating framework-rendered DOM.

### Drop policy events

Drop targeting follows eligibility, priority, then the selected placement
mode. `CanDropEvent` includes `itemMetadata`/`itemsMetadata`, destination
`containerMetadata`, the primary `source`, and `sources` parallel to the full
multi-item drag. `DropPriorityEvent` adds `staticPriority`, pointer and frozen
rectangle geometry, and destination depth. Priority applies only to candidates
owned directly by this container and does not propagate to nested containers.

See [Drop Policies](/docs/snapsort/guides/04_drop_policies) for metadata-driven
grouping, exact resolution rules, and reusable callbacks from
`@snap-engine/snapsort/callbacks`.

## Imperative Ref

Container components forward refs to their core container object.

```tsx
import { useRef } from "react";
import { Container, Item } from "@snap-engine/snapsort/react";
import type { Container as SnapSortContainer } from "@snap-engine/snapsort";

export function ClickMove() {
  const source = useRef<SnapSortContainer>(null);
  const target = useRef<SnapSortContainer>(null);

  function moveToTarget(id: string) {
    if (!source.current || !target.current) return;
    source.current.moveItem(id, target.current, target.current.numberOfItems);
  }

  return (
    <>
      <Container ref={source} config={{ callbacks }}>
        {renderedSourceEntries}
      </Container>
      <Container ref={target} config={{ callbacks }}>
        {renderedTargetEntries}
      </Container>
    </>
  );
}
```

The omitted `callbacks` and rendered-entry construction are the same
state/ghost pattern shown in Setup. Imperative and pointer moves both commit
through `onItemMove`; `moveItem` never authorizes direct DOM mutation. It is an
authoritative application command and bypasses `canDrop` and priority policy.

The ref exposes `moveItem`, `removeItem`, `numberOfItems`, `dropPriority`,
`direction`, `mainAxisAlign`, `mode`, `config`, `callbacks`, and `dragSession`
from the core container.
