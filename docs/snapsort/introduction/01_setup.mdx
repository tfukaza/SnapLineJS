## Install

SnapSort comes in two packages, `@snap-engine/snapsort` itself
and the `@snap-engine/asset-base` dependency.

```bash framework=Svelte
npm install @snap-engine/snapsort @snap-engine/asset-base
```

```bash framework=React
npm install @snap-engine/snapsort @snap-engine/asset-base react react-dom
```

```bash framework=Vanilla
npm install @snap-engine/snapsort
```

## Your First Sortable List

Wrap SnapSort components in an `Engine`, configure the collection callback,
and render one `Item` per entry. This creates a working drag-and-drop list:

```svelte framework=Svelte
<script lang="ts">
  import { Engine } from "@snap-engine/asset-base/svelte";
  import { Container, Ghost, Item } from "@snap-engine/snapsort/svelte";
  import {
    createRenderEntries,
    createRenderTree,
    reduceRenderTree,
    type ContainerCallbacks,
    type GhostLifecycleEvent,
    type ItemMoveEvent,
  } from "@snap-engine/snapsort";

  type Task = { id: string; label: string };

  let list = $state.raw(
    createRenderTree(
      createRenderEntries<Task>(
        [
          { id: "design", label: "Design" },
          { id: "build", label: "Build" },
          { id: "ship", label: "Ship" },
        ],
        (task) => task.id,
      ),
    ),
  );

  function onItemMove(event: ItemMoveEvent) {
    list = reduceRenderTree(list, event);
  }

  function onGhostMove(event: GhostLifecycleEvent) {
    list = reduceRenderTree(list, event);
  }

  const callbacks = {
    onItemMove,
    onGhostInsert: onGhostMove,
    onGhostMove,
    onGhostRemove: onGhostMove,
  } satisfies ContainerCallbacks;
</script>

<Engine id="quick-start">
  <Container itemId="quick-start-root" config={{ callbacks }}>
    {#each list.entries as entry (entry.itemId)}
      {#if entry.isGhost}
        <Ghost ghost={entry.ghost} />
      {:else}
        <Item itemId={entry.itemId}>{entry.value.label}</Item>
      {/if}
    {/each}
  </Container>
</Engine>
```

`RenderTree` is immutable reducer state containing engine-owned identity
objects, so Svelte stores it with `$state.raw` and replaces the root with each
result from `reduceRenderTree`.

`RenderEntry.itemId` is the framework key. SnapSort guarantees that every
ordinary Item, nested Container, and live ghost has a distinct identity inside
one root tree, so no local prefixing scheme is needed.

```tsx framework=React
import { useCallback, useState } from "react";
import { Container, Engine, Ghost, Item } from "@snap-engine/snapsort/react";
import {
  createRenderEntries,
  createRenderTree,
  reduceRenderTree,
} from "@snap-engine/snapsort";
import type { GhostLifecycleEvent, ItemMoveEvent } from "@snap-engine/snapsort";

type Task = { id: string; label: string };

function createInitialList() {
  return createRenderTree(
    createRenderEntries<Task>(
      [
        { id: "design", label: "Design" },
        { id: "build", label: "Build" },
        { id: "ship", label: "Ship" },
      ],
      (task) => task.id,
    ),
  );
}

export function QuickStart() {
  const [list, setList] = useState(createInitialList);
  const onItemMove = useCallback((event: ItemMoveEvent) => {
    setList((current) => reduceRenderTree(current, event));
  }, []);
  const onGhostMove = useCallback((event: GhostLifecycleEvent) => {
    setList((current) => reduceRenderTree(current, event));
  }, []);

  return (
    <Engine id="quick-start">
      <Container
        itemId="quick-start-root"
        config={{
          callbacks: {
            onItemMove,
            onGhostInsert: onGhostMove,
            onGhostMove,
            onGhostRemove: onGhostMove,
          },
        }}
      >
        {list.entries.map((entry) =>
          entry.isGhost ? (
            <Ghost key={entry.itemId} ghost={entry.ghost} />
          ) : (
            <Item key={entry.itemId} itemId={entry.itemId}>
              {entry.value.label}
            </Item>
          ),
        )}
      </Container>
    </Engine>
  );
}
```

## Who Changes Framework Render State?

In Svelte and React, the callbacks above are not a general CRUD layer. They
synchronize changes that **SnapSort originates** with application state. One
semantic item move can remove a complete run from its source and insert it at
its destination, so it produces one root `onItemMove` callback rather than
separate remove and insert notifications.

Ghosts have a different lifetime. SnapSort creates, relocates, and destroys
them while a drag is active, so the framework needs all three explicit
`onGhostInsert`, `onGhostMove`, and `onGhostRemove` hooks. The hooks may still
share one local `GhostLifecycleEvent` function because `reduceRenderTree`
understands all three events.

When application code owns an operation—an Add button, a Delete button, or a
server update—it should update its ordinary `RenderTree` entries directly.
Mounting and unmounting the corresponding framework components registers and
unregisters their core Items; SnapSort does not echo the application's own
change through an item callback.

Add functions like these to the framework examples above:

```ts framework=Svelte
import { createRenderEntry, type ItemId } from "@snap-engine/snapsort";

function addTask(task: Task) {
  list = {
    ...list,
    entries: [...list.entries, createRenderEntry(task, task.id)],
  };
}

function deleteTasks(itemIds: readonly ItemId[]) {
  const deleted = new Set(itemIds);
  list = {
    ...list,
    entries: list.entries.filter(
      (entry) => entry.isGhost || !deleted.has(entry.itemId),
    ),
  };
}
```

```tsx framework=React
import { createRenderEntry, type ItemId } from "@snap-engine/snapsort";

const addTask = useCallback((task: Task) => {
  setList((current) => ({
    ...current,
    entries: [...current.entries, createRenderEntry(task, task.id)],
  }));
}, []);

const deleteTasks = useCallback((itemIds: readonly ItemId[]) => {
  const deleted = new Set(itemIds);
  setList((current) => ({
    ...current,
    entries: current.entries.filter(
      (entry) => entry.isGhost || !deleted.has(entry.itemId),
    ),
  }));
}, []);
```

These flat-list functions are deliberately application code, not SnapSort
exports. A nested application performs the same immutable edit in the selected
child tree. Keep the root Container mounted for the lifetime of this state; if
the root is destroyed, discard the tree and call `createRenderTree` again for
its replacement.

There is one separate imperative path: calling
`container.removeItem(itemId)` changes SnapSort's core tree first. Configure a
root `onItemRemove` and pass that event to `reduceRenderTree` so framework state
catches up synchronously. Ordinary moves never emit that remove callback.
`ItemInsertEvent` is likewise a real lower-level event, but
`reduceRenderTree` cannot accept it because the event cannot invent the
application value to store for a newly introduced core Item.

```javascript framework=Vanilla
import { Engine } from "@snap-engine/core";
import { CollisionEngine } from "@snap-engine/core/collision";
import { Container, Item } from "@snap-engine/snapsort";

const root = document.querySelector("#quick-start");
const list = root.querySelector("[data-snapsort-container]");
const engine = new Engine();

engine.setCollisionEngine(new CollisionEngine());
engine.assignDom(root);

const container = new Container(engine, null, { itemId: "quick-start-root" });
container.element = list;

for (const element of list.querySelectorAll("[data-item-id]")) {
  const itemId = element.dataset.itemId;
  if (!itemId) throw new Error("Missing data-item-id");
  const item = new Item(engine, container, { itemId });
  item.element = element;
}
```

Items can now be dragged to reorder. Animations are opt-in through
`config.animation`; pass `defaultAnimations` for SnapSort's standard motion.
